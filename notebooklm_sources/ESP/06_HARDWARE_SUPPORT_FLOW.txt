#include "OnboardLed.h"
#include "Config.h"
#include "Logger.h"

#include "led_strip.h"
#include "esp_log.h"

static led_strip_handle_t led_strip = nullptr;

void OnboardLed::init() {
    led_strip_config_t strip_config = {
        .strip_gpio_num = PIN_ONBOARD_RGB_LED,
        .max_leds = ONBOARD_RGB_LED_COUNT,
    };

    led_strip_rmt_config_t rmt_config = {
        .resolution_hz = 10 * 1000 * 1000,
    };

    ESP_ERROR_CHECK(led_strip_new_rmt_device(&strip_config, &rmt_config, &led_strip));
    led_strip_clear(led_strip);

    Logger::info("ONBOARD_LED", "Dahili RGB LED baslatildi.");
}

void OnboardLed::setRgb(uint8_t r, uint8_t g, uint8_t b) {
    if (led_strip == nullptr) return;

    led_strip_set_pixel(led_strip, 0, r, g, b);
    led_strip_refresh(led_strip);
}

void OnboardLed::off() {
    if (led_strip == nullptr) return;

    led_strip_clear(led_strip);
}
