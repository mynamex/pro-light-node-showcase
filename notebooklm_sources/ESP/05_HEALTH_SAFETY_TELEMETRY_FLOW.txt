#include "LightRuntimeHealth.h"

#include "esp_log.h"
#include "esp_timer.h"

#define TAG "LIGHT_HEALTH"

static uint32_t s_lastCommandMs = 0;
static uint32_t s_commandAcceptedCount = 0;
static uint32_t s_commandDroppedCount = 0;
static uint32_t s_lastHealthLogMs = 0;

static uint32_t nowMs()
{
    return (uint32_t)(esp_timer_get_time() / 1000ULL);
}

void LightRuntimeHealth::init()
{
    s_lastCommandMs = nowMs();
    s_lastHealthLogMs = nowMs();
    s_commandAcceptedCount = 0;
    s_commandDroppedCount = 0;
    
}

void LightRuntimeHealth::notifyCommandAccepted()
{
    s_lastCommandMs = nowMs();
    s_commandAcceptedCount++;
}

void LightRuntimeHealth::notifyCommandDropped()
{
    s_commandDroppedCount++;
}

uint32_t LightRuntimeHealth::getLastCommandAgeMs()
{
    return nowMs() - s_lastCommandMs;
}

uint32_t LightRuntimeHealth::getCommandAcceptedCount()
{
    return s_commandAcceptedCount;
}

uint32_t LightRuntimeHealth::getCommandDroppedCount()
{
    return s_commandDroppedCount;
}

bool LightRuntimeHealth::hasRecentCommand(uint32_t timeoutMs)
{
    return getLastCommandAgeMs() <= timeoutMs;
}

void LightRuntimeHealth::logPeriodic(uint32_t intervalMs)
{
    uint32_t now = nowMs();

    if ((now - s_lastHealthLogMs) < intervalMs) {
        return;
    }

    s_lastHealthLogMs = now;

    ESP_LOGI(
        TAG,
        "HEALTH | accepted=%lu dropped=%lu lastCommandAgeMs=%lu",
        s_commandAcceptedCount,
        s_commandDroppedCount,
        getLastCommandAgeMs()
    );
}

#include "FanManager.h"

#include "esp_log.h"

#include "FanDriver.h"

#define TAG "FAN_MANAGER"

static FanMode s_mode = FAN_MODE_AUTO;
static uint8_t s_currentFanSpeedPercent = 0;

void FanManager::init()
{
    s_mode = FAN_MODE_AUTO;
    s_currentFanSpeedPercent = 0;

    FanDriver::init();

    ESP_LOGI(TAG, "Fan manager hazir. Mode=AUTO");
}

void FanManager::setMode(FanMode mode)
{
    s_mode = mode;

    ESP_LOGI(TAG, "Fan mode set edildi: %d", (int)mode);
}

FanMode FanManager::getMode()
{
    return s_mode;
}

uint8_t FanManager::getFanSpeedPercent()
{
    return s_currentFanSpeedPercent;
}

uint8_t FanManager::getOutputLimitPercent()
{
    switch (s_mode) {
        case FAN_MODE_SILENT:
            return 40;

        case FAN_MODE_LOW:
            return 60;

        case FAN_MODE_MEDIUM:
            return 80;

        case FAN_MODE_HIGH:
            return 100;

        case FAN_MODE_AUTO:
        default:
            return 100;
    }
}

void FanManager::update()
{
    uint8_t targetSpeed = 0;

    switch (s_mode) {
        case FAN_MODE_SILENT:
            targetSpeed = 0;
            break;

        case FAN_MODE_LOW:
            targetSpeed = 25;
            break;

        case FAN_MODE_MEDIUM:
            targetSpeed = 50;
            break;

        case FAN_MODE_HIGH:
            targetSpeed = 100;
            break;

        case FAN_MODE_AUTO:
        default:
            targetSpeed = 40;
            break;
    }

    if (targetSpeed != s_currentFanSpeedPercent) {
        s_currentFanSpeedPercent = targetSpeed;
        FanDriver::setSpeedPercent(s_currentFanSpeedPercent);
    }
}
