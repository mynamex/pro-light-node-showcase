#include "TimeSyncManager.h"

#include "esp_log.h"
#include "esp_timer.h"

#define TAG "TIME_SYNC"

static bool s_isSynced = false;
static int64_t s_phoneMinusLocalOffsetMs = 0;
static uint32_t s_lastSyncLocalMs = 0;

static uint32_t nowMs()
{
    return (uint32_t)(esp_timer_get_time() / 1000ULL);
}

void TimeSyncManager::init()
{
    s_isSynced = false;
    s_phoneMinusLocalOffsetMs = 0;
    s_lastSyncLocalMs = 0;

    ESP_LOGI(TAG, "Time sync manager hazir.");
}

void TimeSyncManager::updateSync(uint32_t phoneTimeMs)
{
    uint32_t localNow = nowMs();

    s_phoneMinusLocalOffsetMs = (int64_t)phoneTimeMs - (int64_t)localNow;
    s_lastSyncLocalMs = localNow;
    s_isSynced = true;

    ESP_LOGI(
        TAG,
        "SYNC UPDATED | phone=%lu local=%lu offset=%lld",
        (unsigned long)phoneTimeMs,
        (unsigned long)localNow,
        (long long)s_phoneMinusLocalOffsetMs
    );
}

bool TimeSyncManager::isSynced()
{
    return s_isSynced;
}

uint32_t TimeSyncManager::phoneTimeToLocalMs(uint32_t phoneTimeMs)
{
    int64_t local = (int64_t)phoneTimeMs - s_phoneMinusLocalOffsetMs;

    if (local < 0) {
        return 0;
    }

    return (uint32_t)local;
}

uint32_t TimeSyncManager::localTimeToPhoneMs(uint32_t localTimeMs)
{
    int64_t phone = (int64_t)localTimeMs + s_phoneMinusLocalOffsetMs;

    if (phone < 0) {
        return 0;
    }

    return (uint32_t)phone;
}

uint32_t TimeSyncManager::getLastSyncAgeMs()
{
    if (!s_isSynced) {
        return 0xFFFFFFFF;
    }

    return nowMs() - s_lastSyncLocalMs;
}

#include "LightCommandScheduler.h"

#include "esp_log.h"
#include "esp_timer.h"

#include "LightCommandProcessor.h"
#include "LightRuntimeHealth.h"
#include "TimeSyncManager.h"

#define TAG "LIGHT_CMD_SCHED"

static constexpr int MAX_SCHEDULED_COMMANDS = 16;
static constexpr int MAX_RECENT_CUES = 32;

static LightCommand s_scheduledCommands[MAX_SCHEDULED_COMMANDS];
static bool s_scheduledUsed[MAX_SCHEDULED_COMMANDS];

static uint16_t s_recentCueIds[MAX_RECENT_CUES];
static int s_recentCueNext = 0;

static uint32_t nowMs()
{
    return (uint32_t)(esp_timer_get_time() / 1000ULL);
}

void LightCommandScheduler::init()
{
    for (int i = 0; i < MAX_SCHEDULED_COMMANDS; i++) {
        s_scheduledUsed[i] = false;
    }

    for (int i = 0; i < MAX_RECENT_CUES; i++) {
        s_recentCueIds[i] = 0;
    }

    s_recentCueNext = 0;

    ESP_LOGI(TAG, "Light command scheduler hazir.");
}

bool LightCommandScheduler::isDuplicateCue(const LightCommand& command)
{
    if (!command.isV2 || command.cueId == 0) {
        return false;
    }

    for (int i = 0; i < MAX_RECENT_CUES; i++) {
        if (s_recentCueIds[i] == command.cueId) {
            return true;
        }
    }

    s_recentCueIds[s_recentCueNext] = command.cueId;
    s_recentCueNext = (s_recentCueNext + 1) % MAX_RECENT_CUES;

    return false;
}

bool LightCommandScheduler::schedule(const LightCommand& command)
{
    if (isDuplicateCue(command)) {
        ESP_LOGW(
            TAG,
            "Duplicate cue ignore | cueId=%u type=%d value=%u",
            command.cueId,
            (int)command.type,
            command.value
        );
        return true;
    }

     if (command.type == LightCommandType::TIME_SYNC) {
        return LightCommandProcessor::apply(command);
    }

    LightCommand scheduledCommand = command;

    if (!scheduledCommand.isV2 || scheduledCommand.startAtMs == 0) {
        scheduledCommand.applyAtMs = 0;
        return LightCommandProcessor::apply(scheduledCommand);
    }

    if (!TimeSyncManager::isSynced()) {
        ESP_LOGW(
            TAG,
            "Time sync yok, V2 komut hemen uygulanacak | type=%d value=%u startAt=%lu cueId=%u",
            (int)scheduledCommand.type,
            scheduledCommand.value,
            (unsigned long)scheduledCommand.startAtMs,
            scheduledCommand.cueId
        );

        scheduledCommand.applyAtMs = 0;
        return LightCommandProcessor::apply(scheduledCommand);
    }

    scheduledCommand.applyAtMs = TimeSyncManager::phoneTimeToLocalMs(
        scheduledCommand.startAtMs
    );

    if (scheduledCommand.applyAtMs <= nowMs()) {
        ESP_LOGW(
            TAG,
            "Scheduled zaman gecmis, hemen uygulanacak | type=%d value=%u startAt=%lu applyAt=%lu cueId=%u",
            (int)scheduledCommand.type,
            scheduledCommand.value,
            (unsigned long)scheduledCommand.startAtMs,
            (unsigned long)scheduledCommand.applyAtMs,
            scheduledCommand.cueId
        );

        return LightCommandProcessor::apply(scheduledCommand);
    }

    for (int i = 0; i < MAX_SCHEDULED_COMMANDS; i++) {
        if (!s_scheduledUsed[i]) {
            s_scheduledCommands[i] = scheduledCommand;
            s_scheduledUsed[i] = true;

            ESP_LOGI(
                TAG,
                "Scheduled command | slot=%d type=%d value=%u startAt=%lu applyAt=%lu cueId=%u p1=%u p2=%u",
                i,
                (int)scheduledCommand.type,
                scheduledCommand.value,
                (unsigned long)scheduledCommand.startAtMs,
                (unsigned long)scheduledCommand.applyAtMs,
                scheduledCommand.cueId,
                scheduledCommand.param1,
                scheduledCommand.param2
            );

            return true;
        }
    }

    LightRuntimeHealth::notifyCommandDropped();

    ESP_LOGE(
        TAG,
        "Scheduler dolu, komut dusuruldu | type=%d value=%u startAt=%lu cueId=%u",
        (int)scheduledCommand.type,
        scheduledCommand.value,
        (unsigned long)scheduledCommand.startAtMs,
        scheduledCommand.cueId
    );

    return false;
}

void LightCommandScheduler::processDueCommands()
{
    uint32_t now = nowMs();

    for (int i = 0; i < MAX_SCHEDULED_COMMANDS; i++) {
        if (!s_scheduledUsed[i]) {
            continue;
        }

        LightCommand& command = s_scheduledCommands[i];

        if (command.applyAtMs <= now) {
            ESP_LOGI(
                TAG,
                "Apply scheduled command | slot=%d type=%d value=%u cueId=%u",
                i,
                (int)command.type,
                command.value,
                command.cueId
            );

            LightCommandProcessor::apply(command);
            s_scheduledUsed[i] = false;
        }
    }
}

#include "LightCommandQueue.h"
#include "LightRuntimeHealth.h"
#include <freertos/FreeRTOS.h>
#include <freertos/queue.h>

#include "Logger.h"

#define TAG "LIGHT_CMD_QUEUE"

static QueueHandle_t s_lightCommandQueue = nullptr;

bool LightCommandQueue::init()
{
    if (s_lightCommandQueue != nullptr) {
        return true;
    }

    s_lightCommandQueue = xQueueCreate(64, sizeof(LightCommand));

    if (s_lightCommandQueue == nullptr) {
        Logger::error(TAG, "Light command queue olusturulamadi!");
        return false;
    }

    Logger::info(TAG, "Light command queue hazir.");
    return true;
}

bool LightCommandQueue::push(const LightCommand& command)
{
    if (s_lightCommandQueue == nullptr) {
        LightRuntimeHealth::notifyCommandDropped();
        Logger::error(TAG, "Queue hazir degil, komut atilamadi.");
        return false;
    }

    BaseType_t ok = xQueueSend(s_lightCommandQueue, &command, 0);

    if (ok != pdTRUE) {
        LightRuntimeHealth::notifyCommandDropped();
        Logger::error(TAG, "Queue dolu, light komutu dusuruldu.");
        return false;
    }

    return true;
}

bool LightCommandQueue::pop(LightCommand& outCommand)
{
    if (s_lightCommandQueue == nullptr) {
        return false;
    }

    BaseType_t ok = xQueueReceive(s_lightCommandQueue, &outCommand, 0);
    return ok == pdTRUE;
}

bool LightCommandQueue::isReady()
{
    return s_lightCommandQueue != nullptr;
}
