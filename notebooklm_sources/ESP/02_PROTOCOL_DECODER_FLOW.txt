#include "ProLightProtocolDecoder.h"

#include <string.h>

static ProLightCommandId mapLegacyCommand(char command)
{
    switch (command) {
        case 'R':
            return PRO_LIGHT_CMD_RED;

        case 'G':
            return PRO_LIGHT_CMD_GREEN;

        case 'B':
            return PRO_LIGHT_CMD_BLUE;

        case 'M':
            return PRO_LIGHT_CMD_MASTER;

        case 'E':
            return PRO_LIGHT_CMD_EFFECT;

        case 'F':
            return PRO_LIGHT_CMD_FAN;

        case 'P':
            return PRO_LIGHT_CMD_PING;

        case 'H':
            return PRO_LIGHT_CMD_HEALTH;

        default:
            return PRO_LIGHT_CMD_UNKNOWN;
    }
}

bool pro_light_decode_payload(
    const uint8_t *payload,
    uint16_t length,
    ProLightDecodedCommand *outCommand
) {
    if (outCommand == nullptr) {
        return false;
    }

    memset(outCommand, 0, sizeof(ProLightDecodedCommand));

    if (payload == nullptr) {
        return false;
    }

    if (length == 3) {
        char legacyCommand = (char)payload[0];
        ProLightCommandId commandId = mapLegacyCommand(legacyCommand);

        if (commandId == PRO_LIGHT_CMD_UNKNOWN) {
            return false;
        }

        outCommand->isValid = true;
        outCommand->isV2 = false;
        outCommand->commandId = commandId;
        outCommand->value = payload[1] | (payload[2] << 8);
        outCommand->startAtMs = 0;
        outCommand->cueId = 0;
        outCommand->param1 = 0;
        outCommand->param2 = 0;

        return true;
    }

    if (length >= PRO_LIGHT_PROTOCOL_V2_LEN &&
        payload[0] == PRO_LIGHT_PROTOCOL_V2_VERSION) {

        ProLightCommandId commandId = (ProLightCommandId)payload[1];

        if (commandId == PRO_LIGHT_CMD_UNKNOWN ||
            commandId > PRO_LIGHT_CMD_TIME_SYNC) {
            return false;
        }

        outCommand->isValid = true;
        outCommand->isV2 = true;
        outCommand->commandId = commandId;
        outCommand->value = payload[2] | (payload[3] << 8);

        outCommand->startAtMs =
            ((uint32_t)payload[4]) |
            ((uint32_t)payload[5] << 8) |
            ((uint32_t)payload[6] << 16) |
            ((uint32_t)payload[7] << 24);

        outCommand->cueId = payload[8] | (payload[9] << 8);
        outCommand->param1 = payload[10];
        outCommand->param2 = payload[11];

        return true;
    }

    return false;
}

#include "node_mesh_vendor.h"

#include <stdio.h>
#include <stdbool.h>

#include "esp_log.h"
#include "esp_mac.h"
#include "esp_ble_mesh_networking_api.h"

#include "node_mesh.h"
#include "node_mesh_defs.h"
#include "TelemetryBridge.h"
#include "ProLightProtocolDecoder.h"

#define TAG "MESH_VENDOR"

static esp_ble_mesh_model_t *s_vendor_model = NULL;

void node_mesh_vendor_set_model(esp_ble_mesh_model_t *model)
{
    s_vendor_model = model;
}

static void fill_status_payload(uint8_t out[3], char type, uint16_t value)
{
    out[0] = (uint8_t)type;
    out[1] = value & 0xFF;
    out[2] = (value >> 8) & 0xFF;
}

static bool is_unicast_addr(uint16_t addr)
{
    return addr >= 0x0001 && addr <= 0x7FFF;
}

static bool is_group_addr(uint16_t addr)
{
    return addr >= 0xC000 && addr <= 0xFEFF;
}

static char command_id_to_legacy_char(ProLightCommandId commandId)
{
    switch (commandId) {
        case PRO_LIGHT_CMD_RED:
            return 'R';

        case PRO_LIGHT_CMD_GREEN:
            return 'G';

        case PRO_LIGHT_CMD_BLUE:
            return 'B';

        case PRO_LIGHT_CMD_MASTER:
            return 'M';

        case PRO_LIGHT_CMD_EFFECT:
            return 'E';

        case PRO_LIGHT_CMD_FAN:
            return 'F';

        case PRO_LIGHT_CMD_PING:
            return 'P';

        case PRO_LIGHT_CMD_HEALTH:
            return 'H';
        case PRO_LIGHT_CMD_TIME_SYNC:
            return 'T';

        default:
            return '?';
    }
}

static void send_status_response(
    esp_ble_mesh_model_cb_param_t *param,
    char type,
    uint16_t value
) {
    uint8_t mac[6];
    esp_read_mac(mac, ESP_MAC_BT);

    char mac_str[18];
    snprintf(
        mac_str,
        sizeof(mac_str),
        "%02X:%02X:%02X:%02X:%02X:%02X",
        mac[0], mac[1], mac[2],
        mac[3], mac[4], mac[5]
    );

    uint16_t self_addr = param->model_operation.model->element->element_addr;

    uint8_t status_payload[3];
    fill_status_payload(status_payload, type, value);

    esp_err_t err = esp_ble_mesh_server_model_send_msg(
        param->model_operation.model,
        param->model_operation.ctx,
        PRO_LIGHT_OP_STATUS,
        sizeof(status_payload),
        status_payload
    );

    if (err == ESP_OK) {
        ESP_LOGI(
            TAG,
            "STATUS RESPONSE | MAC=%s SELF=0x%04x -> DST=0x%04x TYPE=%c VALUE=%u",
            mac_str,
            self_addr,
            param->model_operation.ctx->addr,
            type,
            value
        );
    } else {
        ESP_LOGE(
            TAG,
            "STATUS RESPONSE hatasi | MAC=%s SELF=0x%04x err=%d",
            mac_str,
            self_addr,
            err
        );
    }
}

void node_mesh_vendor_model_cb(
    esp_ble_mesh_model_cb_event_t event,
    esp_ble_mesh_model_cb_param_t *param
) {
    if (event != ESP_BLE_MESH_MODEL_OPERATION_EVT) {
        return;
    }

    if (param == NULL || param->model_operation.ctx == NULL) {
        ESP_LOGE(TAG, "MESH_VENDOR RX param veya ctx NULL");
        return;
    }

    uint8_t mac[6];
    esp_read_mac(mac, ESP_MAC_BT);

    char mac_str[18];
    snprintf(
        mac_str,
        sizeof(mac_str),
        "%02X:%02X:%02X:%02X:%02X:%02X",
        mac[0], mac[1], mac[2],
        mac[3], mac[4], mac[5]
    );

    uint32_t opcode = param->model_operation.opcode;
    uint16_t len = param->model_operation.length;
    uint8_t *msg = param->model_operation.msg;

    uint16_t self_addr = param->model_operation.model->element->element_addr;
    uint16_t src_addr = param->model_operation.ctx->addr;
    uint16_t dst_addr = param->model_operation.ctx->recv_dst;
    uint8_t recv_ttl = param->model_operation.ctx->recv_ttl;

    if (msg == NULL) {
        ESP_LOGE(TAG, "Payload NULL | SRC=0x%04x DST=0x%04x", src_addr, dst_addr);
        return;
    }

    ESP_LOGI(
        TAG,
        "MESH_VENDOR RX | MAC=%s SELF=0x%04x SRC=0x%04x DST=0x%04x TTL=%u OPCODE=0x%06x LEN=%u",
        mac_str,
        self_addr,
        src_addr,
        dst_addr,
        recv_ttl,
        (unsigned int)opcode,
        len
    );

    if (opcode != PRO_LIGHT_OP_SEND) {
        ESP_LOGW(
            TAG,
            "Bilinmeyen opcode reddedildi | OPCODE=0x%06x SRC=0x%04x DST=0x%04x",
            (unsigned int)opcode,
            src_addr,
            dst_addr
        );
        return;
    }

    ProLightDecodedCommand decoded;

    if (!pro_light_decode_payload(msg, len, &decoded)) {
        ESP_LOGW(
            TAG,
            "Payload decode edilemedi | MAC=%s SELF=0x%04x SRC=0x%04x DST=0x%04x LEN=%u",
            mac_str,
            self_addr,
            src_addr,
            dst_addr,
            len
        );
        return;
    }

    char command = command_id_to_legacy_char(decoded.commandId);

    if (command == '?') {
        ESP_LOGW(TAG, "Bilinmeyen decoded commandId=%u", decoded.commandId);
        return;
    }

    uint16_t value = decoded.value;

    if (value > 1023 &&
        decoded.commandId != PRO_LIGHT_CMD_PING &&
        decoded.commandId != PRO_LIGHT_CMD_HEALTH) {
        ESP_LOGW(
            TAG,
            "Value clamp uygulandi | CMD=%c VALUE=%u -> 1023",
            command,
            value
        );
        value = 1023;
    }

    bool dst_is_unicast = is_unicast_addr(dst_addr);
    bool dst_is_group = is_group_addr(dst_addr);

    ESP_LOGI(
        TAG,
        "MESH_VENDOR HIT | MAC=%s SELF=0x%04x SRC=0x%04x DST=0x%04x TTL=%u CMD=%c VALUE=%u isGroup=%d V2=%d startAtMs=%lu cueId=%u p1=%u p2=%u",
        mac_str,
        self_addr,
        src_addr,
        dst_addr,
        recv_ttl,
        command,
        value,
        dst_is_group,
        decoded.isV2,
        (unsigned long)decoded.startAtMs,
        decoded.cueId,
        decoded.param1,
        decoded.param2
    );

    if (decoded.commandId == PRO_LIGHT_CMD_HEALTH) {
        if (!dst_is_unicast) {
            ESP_LOGW(
                TAG,
                "HEALTH grup adresine geldi, cevap donulmedi | MAC=%s SELF=0x%04x DST=0x%04x TTL=%u",
                mac_str,
                self_addr,
                dst_addr,
                recv_ttl
            );
            return;
        }

        switch (value) {
            case 1:
                send_status_response(param, 't', telemetry_get_temperature_x10());
                break;

            case 2:
                send_status_response(param, 'f', telemetry_get_fan_mode());
                break;

            case 3:
                send_status_response(param, 'v', telemetry_get_fan_speed_percent());
                break;

            case 4:
                send_status_response(param, 'h', telemetry_get_thermal_status());
                break;

            case 5:
                send_status_response(param, 'd', telemetry_get_command_dropped_count());
                break;

            case 6:
                send_status_response(param, 'a', telemetry_get_command_accepted_count());
                break;

            default:
                ESP_LOGW(
                    TAG,
                    "Bilinmeyen HEALTH request | value=%u MAC=%s SELF=0x%04x",
                    value,
                    mac_str,
                    self_addr
                );
                send_status_response(param, 'h', 0xFFFF);
                break;
        }

        return;
    }

    if (decoded.commandId == PRO_LIGHT_CMD_PING) {
        ESP_LOGI(
            TAG,
            "PING geldi | MAC=%s SELF=0x%04x SRC=0x%04x DST=0x%04x TTL=%u",
            mac_str,
            self_addr,
            src_addr,
            dst_addr,
            recv_ttl
        );

        if (dst_is_unicast) {
            send_status_response(param, 'S', 1);
        } else {
            ESP_LOGW(
                TAG,
                "PING grup adresine geldi, cevap donulmedi | MAC=%s SELF=0x%04x DST=0x%04x TTL=%u",
                mac_str,
                self_addr,
                dst_addr,
                recv_ttl
            );
        }

        return;
    }

on_mesh_command_received(
    command,
    value,
    src_addr,
    dst_addr,
    self_addr,
    recv_ttl,
    decoded.isV2,
    decoded.startAtMs,
    decoded.cueId,
    decoded.param1,
    decoded.param2
);
}
