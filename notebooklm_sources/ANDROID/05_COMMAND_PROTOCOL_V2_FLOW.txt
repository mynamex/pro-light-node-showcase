    private val meshController: MeshController,

    private val onStatus: (String) -> Unit
) {
    private val handler = Handler(Looper.getMainLooper())

    fun sendPower(groupAddress: Int, turnOn: Boolean) {
        if (!isProxyReady()) return

        val value = if (turnOn) 1023 else 0
        onStatus(if (turnOn) "Işık açılıyor..." else "Işık kapatılıyor...")

        sendGroupDim(groupAddress, value)
    }

    fun sendRgb(groupAddress: Int, r: Int, g: Int, b: Int) {
        if (!isProxyReady()) return

        onStatus("RGB GROUP gönderiliyor -> group=0x${groupAddress.toString(16)} R:$r G:$g B:$b")

        meshCommander.sendGroupCommand(groupAddress, 'R', r)

        handler.postDelayed({
            if (isProxyReady()) {
                meshCommander.sendGroupCommand(groupAddress, 'G', g)
            }
        }, 500L)

        handler.postDelayed({
            if (isProxyReady()) {
                meshCommander.sendGroupCommand(groupAddress, 'B', b)
            }
        }, 1000L)

        handler.postDelayed({
            if (isProxyReady()) {
                meshCommander.sendGroupCommand(groupAddress, 'M', 1023)
            }
        }, 1500L)
    }

package com.example.prolightmesh.mesh

import android.util.Log
import com.example.prolightmesh.mesh.protocol.ProLightCommandIds
import com.example.prolightmesh.mesh.protocol.ProLightCueManager
import com.example.prolightmesh.mesh.protocol.ProLightV2PacketBuilder
import no.nordicsemi.android.mesh.transport.VendorModelMessageAcked
import no.nordicsemi.android.mesh.transport.VendorModelMessageUnacked
import java.nio.ByteBuffer
import java.nio.ByteOrder

class MeshCommander(private val meshController: MeshController) {

    companion object {
        private const val TAG = "MESH_COMMANDER"

        private const val COMPANY_ID = 0x02E5
        private const val VENDOR_MODEL_ID = 0x0001
        private const val OPCODE_SEND = 0x00

        private const val GROUP_ADDRESS_START = 0xC000
        private const val GROUP_ADDRESS_END = 0xFEFF

        const val EFFECT_NONE = 0
        const val EFFECT_POLICE = 1
        const val EFFECT_AMBULANCE = 2
        const val EFFECT_FIRE = 3
        const val EFFECT_LIGHTNING = 4

        const val FAN_SILENT = 0
        const val FAN_LOW = 1
        const val FAN_MEDIUM = 2
        const val FAN_HIGH = 3
        const val FAN_AUTO = 4

        const val HEALTH_TEMPERATURE_X10 = 1
        const val HEALTH_FAN_MODE = 2
        const val HEALTH_FAN_SPEED_PERCENT = 3
        const val HEALTH_THERMAL_STATUS = 4
        const val HEALTH_DROPPED_COUNT = 5
        const val HEALTH_ACCEPTED_COUNT = 6
    }

    fun sendProLightCommand(dstAddress: Int, commandType: Char, value: Int) {
        if (isGroupAddress(dstAddress)) {
            sendGroupCommand(dstAddress, commandType, value)
        } else {
            sendNodeCommandAcked(dstAddress, commandType, value)
        }
    }

    fun sendPing(nodeAddress: Int) {
        sendNodeCommandUnacked(nodeAddress, 'P', 0)
    }

    fun sendEffect(dstAddress: Int, effectValue: Int) {
        sendProLightCommand(dstAddress, 'E', effectValue)
    }

    fun sendFanMode(dstAddress: Int, fanMode: Int) {
        sendProLightCommand(dstAddress, 'F', fanMode)
    }

    fun sendHealthRequest(nodeAddress: Int, requestType: Int) {
        sendNodeCommandUnacked(nodeAddress, 'H', requestType)
    }

    fun sendTimeSync(nodeAddress: Int) {
        val phoneNow32 = ProLightV2PacketBuilder.phoneNow32()

        val payload = ProLightV2PacketBuilder.buildV2Packet(
            commandId = ProLightCommandIds.TIME_SYNC,
            value = 0,
            startAtMs = phoneNow32,
            cueId = 0,
            param1 = 0,
            param2 = 0
        )

        Log.w(
            "TIME_SYNC",
            "TX -> node=0x${nodeAddress.toString(16)} phoneNow32=$phoneNow32"
        )

        sendNodePayloadUnacked(nodeAddress, payload)
    }

    fun sendScheduledGroupEffect(
        groupAddress: Int,
        effectValue: Int,
        delayMs: Long = 500L,
        speed: Int = 0,
        intensity: Int = 0
    ) {
        val now32 = ProLightV2PacketBuilder.phoneNow32()
        val startAt32 = (now32 + delayMs) and 0xFFFFFFFFL
        val cueId = ProLightCueManager.next()

        val payload = ProLightV2PacketBuilder.buildV2Packet(
            commandId = ProLightCommandIds.EFFECT,
            value = effectValue,
            startAtMs = startAt32,
            cueId = cueId,
            param1 = speed,
            param2 = intensity
        )

        Log.w(
            "V2_SCHEDULE",
            "GROUP EFFECT TX -> group=0x${groupAddress.toString(16)} effect=$effectValue startAt32=$startAt32 cueId=$cueId speed=$speed intensity=$intensity"
        )

        sendGroupPayloadUnacked(groupAddress, payload)
    }

    fun sendNodeCommandUnacked(dstAddress: Int, commandType: Char, value: Int) {
        val payload = buildLegacyPayload(commandType, value)

        Log.d(
            TAG,
            "📨 Vendor UNACKED NODE TX -> dst=0x${dstAddress.toString(16)}, type=$commandType, value=$value"
        )

        sendNodePayloadUnacked(dstAddress, payload)
    }

    fun sendNodeCommandAcked(dstAddress: Int, commandType: Char, value: Int) {
        val network = meshController.meshManagerApi.meshNetwork ?: run {
            Log.e(TAG, "HATA: Mesh network yok")
            return
        }

        val appKey = network.appKeys.firstOrNull() ?: run {
            Log.e(TAG, "HATA: AppKey yok")
            return
        }

        val payload = buildLegacyPayload(commandType, value)

        Log.d(
            TAG,
            "🚀 Vendor ACKED NODE TX -> dst=0x${dstAddress.toString(16)}, type=$commandType, value=$value"
        )

        val message = VendorModelMessageAcked(
            appKey,
            VENDOR_MODEL_ID,
            COMPANY_ID,
            OPCODE_SEND,
            payload
        )

        meshController.sendMeshMessage(dstAddress, message)
    }

    fun sendGroupCommand(dstAddress: Int, commandType: Char, value: Int) {
        val payload = buildLegacyPayload(commandType, value)

        Log.d(
            TAG,
            "📡 Vendor UNACKED GROUP TX -> dst=0x${dstAddress.toString(16)}, type=$commandType, value=$value"
        )

        sendGroupPayloadUnacked(dstAddress, payload)
    }

    private fun sendNodePayloadUnacked(dstAddress: Int, payload: ByteArray) {
        val network = meshController.meshManagerApi.meshNetwork ?: run {
            Log.e(TAG, "HATA: Mesh network yok")
            return
        }

        val appKey = network.appKeys.firstOrNull() ?: run {
            Log.e(TAG, "HATA: AppKey yok")
            return
        }

        val message = VendorModelMessageUnacked(
            appKey,
            VENDOR_MODEL_ID,
            COMPANY_ID,
            OPCODE_SEND,
            payload
        )

        meshController.sendMeshMessage(dstAddress, message)
    }

    private fun sendGroupPayloadUnacked(dstAddress: Int, payload: ByteArray) {
        val network = meshController.meshManagerApi.meshNetwork ?: run {
            Log.e(TAG, "HATA: Mesh network yok")
            return
        }

        val appKey = network.appKeys.firstOrNull() ?: run {
            Log.e(TAG, "HATA: AppKey yok")
            return
        }

        val message = VendorModelMessageUnacked(
            appKey,
            VENDOR_MODEL_ID,
            COMPANY_ID,
            OPCODE_SEND,
            payload
        )

        meshController.sendMeshMessage(dstAddress, message)
    }

    private fun buildLegacyPayload(commandType: Char, value: Int): ByteArray {
        return ByteBuffer.allocate(3)
            .order(ByteOrder.LITTLE_ENDIAN)
            .put(commandType.code.toByte())
            .putShort(value.toShort())
            .array()
    }

    private fun isGroupAddress(address: Int): Boolean {
        return address in GROUP_ADDRESS_START..GROUP_ADDRESS_END
    }
}

package com.example.prolightmesh.mesh.protocol

import java.nio.ByteBuffer
import java.nio.ByteOrder

object ProLightV2PacketBuilder {

    private const val VERSION_V2 = 2
    private const val PACKET_SIZE = 12

    fun buildV2Packet(
        commandId: Int,
        value: Int,
        startAtMs: Long,
        cueId: Int,
        param1: Int = 0,
        param2: Int = 0
    ): ByteArray {
        return ByteBuffer.allocate(PACKET_SIZE)
            .order(ByteOrder.LITTLE_ENDIAN)
            .put(VERSION_V2.toByte())
            .put(commandId.toByte())
            .putShort((value and 0xFFFF).toShort())
            .putInt((startAtMs and 0xFFFFFFFFL).toInt())
            .putShort((cueId and 0xFFFF).toShort())
            .put((param1 and 0xFF).toByte())
            .put((param2 and 0xFF).toByte())
            .array()
    }

    fun phoneNow32(): Long {
        return System.currentTimeMillis() and 0xFFFFFFFFL
    }
}

object ProLightCueManager {

    private var nextCueId: Int = 1

    @Synchronized
    fun next(): Int {
        val id = nextCueId

        nextCueId++

        if (nextCueId > 65000) {
            nextCueId = 1
        }

        return id
    }
}

package com.example.prolightmesh.mesh.protocol

import android.os.Handler
import android.os.SystemClock
import android.util.Log
import com.example.prolightmesh.mesh.MeshCommander
import com.example.prolightmesh.mesh.MeshController

class TimeSyncService(
    private val meshController: MeshController,
    private val meshCommander: MeshCommander,
    private val mainHandler: Handler,
    private val onStatus: (String) -> Unit
) {
    private var lastSyncStartedAtMs: Long = 0L
    private var lastSyncFinishedAtMs: Long = 0L

    fun ensureFreshSync(
        maxAgeMs: Long = DEFAULT_MAX_SYNC_AGE_MS,
        onReady: () -> Unit
    ) {
        val age = getLastSyncAgeMs()

        if (lastSyncFinishedAtMs > 0L && age <= maxAgeMs) {
            Log.w(TAG, "Sync fresh. ageMs=$age")
            onReady()
            return
        }

        Log.w(TAG, "Sync old/missing. ageMs=$age -> syncing all nodes")

        syncAllKnownNodes(
            onComplete = {
                Log.w(TAG, "Sync complete. Running scheduled action.")
                onReady()
            }
        )
    }

    fun syncAllKnownNodes(onComplete: (() -> Unit)? = null) {
        if (!meshController.isProxyReady()) {
            val msg = "TIME_SYNC yapılamadı: Proxy hazır değil."
            Log.e(TAG, msg)
            onStatus(msg)
            return
        }

        val network = meshController.meshManagerApi.meshNetwork ?: run {
            val msg = "TIME_SYNC yapılamadı: Mesh network yok."
            Log.e(TAG, msg)
            onStatus(msg)
            return
        }

        val provisionerAddress = network.provisionerAddress

        val nodes = network.nodes
            .filter { it.unicastAddress != provisionerAddress }

        if (nodes.isEmpty()) {
            val msg = "TIME_SYNC: sync edilecek node yok."
            Log.w(TAG, msg)
            onStatus(msg)
            onComplete?.invoke()
            return
        }

        lastSyncStartedAtMs = SystemClock.elapsedRealtime()

        onStatus("TIME_SYNC başlıyor -> node count=${nodes.size}")

        var delay = 0L

        nodes.forEachIndexed { index, node ->
            mainHandler.postDelayed({
                if (!meshController.isProxyReady()) {
                    Log.e(
                        TAG,
                        "TIME_SYNC skip: proxy hazır değil. node=0x${node.unicastAddress.toString(16)}"
                    )
                    return@postDelayed
                }

                Log.w(
                    TAG,
                    "TIME_SYNC node ${index + 1}/${nodes.size} -> 0x${node.unicastAddress.toString(16)}"
                )

                meshCommander.sendTimeSync(node.unicastAddress)
            }, delay)

            delay += SYNC_NODE_GAP_MS
        }

        mainHandler.postDelayed({
            lastSyncFinishedAtMs = SystemClock.elapsedRealtime()

            val duration = lastSyncFinishedAtMs - lastSyncStartedAtMs

            Log.w(TAG, "TIME_SYNC finished. durationMs=$duration")
            onStatus("TIME_SYNC tamamlandı. duration=${duration}ms")

            onComplete?.invoke()
        }, delay + SYNC_SETTLE_MS)
    }

    fun getLastSyncAgeMs(): Long {
        if (lastSyncFinishedAtMs == 0L) {
            return Long.MAX_VALUE
        }

        return SystemClock.elapsedRealtime() - lastSyncFinishedAtMs
    }

    fun markSyncInvalid(reason: String) {
        Log.w(TAG, "Sync invalidated. reason=$reason")
        lastSyncFinishedAtMs = 0L
    }

    companion object {
        private const val TAG = "TIME_SYNC_SERVICE"

        private const val DEFAULT_MAX_SYNC_AGE_MS = 10_000L
        private const val SYNC_NODE_GAP_MS = 250L
        private const val SYNC_SETTLE_MS = 300L
    }
}

package com.example.prolightmesh.ui.work

import android.app.Activity
import android.app.AlertDialog
import android.graphics.Color
import android.graphics.Typeface
import android.graphics.drawable.ColorDrawable
import android.graphics.drawable.GradientDrawable
import android.view.Gravity
import android.view.ViewGroup
import android.view.Window
import android.widget.LinearLayout
import android.widget.ScrollView
import android.widget.TextView
import com.example.prolightmesh.storage.StoredGroup
import com.example.prolightmesh.ui.theme.ProColors

class WorkGroupControlDialog(
    private val activity: Activity,
    private val group: StoredGroup,
    private val onStatus: (String) -> Unit,
    private val onGroupColorChanged: (Int, Int, Int, Int) -> Unit,
    private val onGroupDimChanged: (Int, Int) -> Unit,
    private val onGroupEffectChanged: (Int, Int) -> Unit,
    private val onGroupFanModeChanged: (Int, Int) -> Unit
) {
    private lateinit var dialog: AlertDialog
    private lateinit var contentContainer: LinearLayout
    private lateinit var tabSelector: WorkTabSelector

    fun show() {
        val root = LinearLayout(activity).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(dp(20), dp(22), dp(20), dp(20))
            background = roundedBackground(ProColors.Surface, dp(28))
        }

        root.addView(headerView())

        tabSelector = WorkTabSelector(
            context = activity,
            tabs = listOf("IŞIKLAR", "RENK", "EFFECT", "FAN"),
            onTabSelected = { index -> showTab(index) }
        )

        root.addView(tabSelector.rootView)

        contentContainer = LinearLayout(activity).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(0, dp(20), 0, 0)
        }

        val scrollView = ScrollView(activity).apply {
            isFillViewport = false
            addView(contentContainer)
        }

        root.addView(
            scrollView,
            LinearLayout.LayoutParams(
                ViewGroup.LayoutParams.MATCH_PARENT,
                dp(560)
            )
        )

        dialog = AlertDialog.Builder(activity)
            .setView(root)
            .create()

        dialog.requestWindowFeature(Window.FEATURE_NO_TITLE)
        dialog.window?.setBackgroundDrawable(ColorDrawable(Color.TRANSPARENT))
        dialog.show()

        dialog.window?.setLayout(
            (activity.resources.displayMetrics.widthPixels * 0.94f).toInt(),
            ViewGroup.LayoutParams.WRAP_CONTENT
        )

        showTab(0)
    }

    private fun showTab(index: Int) {
        contentContainer.removeAllViews()
        tabSelector.setSelected(index)

        val view = when (index) {
            0 -> WorkLightTabView.create(
                context = activity,
                group = group,
                onStatus = onStatus,
                onGroupColorChanged = onGroupColorChanged,
                onGroupDimChanged = onGroupDimChanged
            )

            1 -> WorkColorTabView.create(
                context = activity,
                group = group,
                onStatus = onStatus,
                onGroupColorChanged = onGroupColorChanged
            )

            2 -> WorkEffectTabView.create(
                context = activity,
                group = group,
                onStatus = onStatus,
                onGroupEffectChanged = onGroupEffectChanged
            )

            else -> WorkFanTabView.create(
                context = activity,
                group = group,
                onStatus = onStatus,
                onGroupFanModeChanged = onGroupFanModeChanged
            )
        }

        contentContainer.addView(view)
    }

    private fun headerView(): LinearLayout {
        return LinearLayout(activity).apply {
            orientation = LinearLayout.VERTICAL
            gravity = Gravity.CENTER_HORIZONTAL
            setPadding(0, 0, 0, dp(18))

            addView(TextView(activity).apply {
                text = group.name
                textSize = 22f
                setTypeface(null, Typeface.BOLD)
                setTextColor(ProColors.TextPrimary)
                gravity = Gravity.CENTER
            })

            addView(TextView(activity).apply {
                text = "0x${group.address.toString(16)} • Grup Kontrol Paneli"
                textSize = 13f
                setTextColor(ProColors.TextSecondary)
                gravity = Gravity.CENTER
                setPadding(0, dp(6), 0, 0)
            })
        }
    }

    private fun roundedBackground(color: Int, radius: Int): GradientDrawable {
        return GradientDrawable().apply {
            setColor(color)
            cornerRadius = radius.toFloat()
        }
    }

    private fun dp(value: Int): Int {
        return (value * activity.resources.displayMetrics.density).toInt()
    }
}
