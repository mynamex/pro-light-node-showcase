package com.example.prolightmesh.ui.navigation

import android.app.Activity
import android.graphics.Color
import android.view.View
import android.widget.FrameLayout
import android.widget.LinearLayout
import com.example.prolightmesh.ui.screens.GroupsScreen
import com.example.prolightmesh.ui.screens.NetworkScreen
import com.example.prolightmesh.ui.screens.NewDeviceScreen
import com.example.prolightmesh.ui.screens.WorkScreen

class AppNavigator(
    private val activity: Activity,
    private val bottomNavView: BottomNavView,
    private val workScreen: WorkScreen,
    private val networkScreen: NetworkScreen,
    private val groupsScreen: GroupsScreen,
    private val newDeviceScreen: NewDeviceScreen // YENİ: ScanView yerine NewDeviceScreen geldi
) {

    enum class Screen {
        GROUPS,
        NETWORK,
        WORK,
        NEW_DEVICE
        // DETAIL silindi, çünkü artık diyalog kullanıyoruz
    }

    var currentScreen: Screen = Screen.WORK
        private set

    fun showWork() {
        currentScreen = Screen.WORK
        showWithBottomNav(workScreen.rootView)
    }

    fun showGroups() {
        currentScreen = Screen.GROUPS
        showWithBottomNav(groupsScreen.rootView)
    }

    fun showNetwork() {
        currentScreen = Screen.NETWORK
        showWithBottomNav(networkScreen.rootView)
    }

    fun showNewDevice() {
        currentScreen = Screen.NEW_DEVICE
        showWithBottomNav(newDeviceScreen.rootView)
    }

    private fun showWithBottomNav(contentView: View) {
        val root = LinearLayout(activity).apply {
            orientation = LinearLayout.VERTICAL
            setBackgroundColor(Color.parseColor("#121212")) // Zifiri antrasit arkaplan
        }

        val contentContainer = FrameLayout(activity).apply {
            (contentView.parent as? android.view.ViewGroup)?.removeView(contentView)
            addView(contentView)
        }

        root.addView(
            contentContainer,
            LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                0,
                1f
            )
        )

        (bottomNavView.rootView.parent as? android.view.ViewGroup)
            ?.removeView(bottomNavView.rootView)

        root.addView(
            bottomNavView.rootView,
            LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            )
        )

        activity.setContentView(root)
    }
}

class GroupsScreen(
    private val activity: Activity,
    private val onAddGroup: (String) -> Unit,
    private val onDeleteGroup: (StoredGroup) -> Unit
) {
    val rootView: ScrollView

    private val container: LinearLayout
    private val statusText: TextView

    init {
        container = LinearLayout(activity).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(32, 48, 32, 32)
            setBackgroundColor(Color.rgb(20, 20, 20))
        }

        val title = TextView(activity).apply {
            text = "GRUPLAR"
            textSize = 26f
            setTypeface(null, Typeface.BOLD)
            setTextColor(Color.WHITE)
            gravity = Gravity.CENTER
            setPadding(0, 0, 0, 12)
        }

        statusText = TextView(activity).apply {
            text = "Işık grupları"
            textSize = 14f
            setTextColor(Color.GREEN)
            gravity = Gravity.CENTER
            setPadding(0, 0, 0, 24)
        }

        val addButton = Button(activity).apply {
            text = "➕ GRUP OLUŞTUR"
            textSize = 16f
            setTextColor(Color.WHITE)
            setBackgroundColor(Color.rgb(60, 90, 150))
            setPadding(0, 28, 0, 28)
            setOnClickListener { showAddGroupDialog() }
        }

        container.addView(title)
        container.addView(statusText)
        container.addView(addButton)

        rootView = ScrollView(activity).apply {
            isFillViewport = true
            addView(container)
        }
    }

    fun renderGroups(groups: List<StoredGroup>) {
        clearGroupRows()

        statusText.text = "${groups.size} grup"

        groups.forEach { group ->
            container.addView(groupRow(group))
        }
    }

    private fun groupRow(group: StoredGroup): LinearLayout {
        return LinearLayout(activity).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(24, 24, 24, 24)
            setBackgroundColor(Color.rgb(35, 35, 35))

            val title = TextView(activity).apply {
                text = group.name
                textSize = 18f
                setTypeface(null, Typeface.BOLD)
                setTextColor(Color.WHITE)
            }

            val subtitle = TextView(activity).apply {
                text = "Adres: 0x${group.address.toString(16)}"
                textSize = 14f
                setTextColor(Color.LTGRAY)
                setPadding(0, 6, 0, 12)
            }

            val deleteButton = Button(activity).apply {
                text = "🗑 SİL"
                setTextColor(Color.WHITE)
                setBackgroundColor(Color.rgb(130, 0, 0))
                setOnClickListener { onDeleteGroup(group) }

                isEnabled = group.address != 0xC000
                alpha = if (group.address == 0xC000) 0.4f else 1f
            }

            addView(title)
            addView(subtitle)
            addView(deleteButton)

            layoutParams = LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            ).apply {
                setMargins(0, 24, 0, 0)
            }
        }
    }

    private fun clearGroupRows() {
        while (container.childCount > 3) {
            container.removeViewAt(3)
        }
    }

    private fun showAddGroupDialog() {
        val input = EditText(activity).apply {
            hint = "Örn: Key Light"
            inputType = InputType.TYPE_CLASS_TEXT
        }

        AlertDialog.Builder(activity)
            .setTitle("Yeni Grup")
            .setMessage("Grup adını gir:")
            .setView(input)
            .setPositiveButton("Oluştur") { _, _ ->
                val name = input.text.toString().trim()
                if (name.isNotBlank()) {
                    onAddGroup(name)
                }
            }
            .setNegativeButton("İptal", null)
            .show()
    }
}

package com.example.prolightmesh.ui.components

import android.annotation.SuppressLint
import android.content.Context
import android.graphics.*
import android.view.MotionEvent
import android.view.View
import com.example.prolightmesh.ui.theme.ProColors
import kotlin.math.atan2
import kotlin.math.cos
import kotlin.math.hypot
import kotlin.math.sin

class ColorWheelPanel(
    context: Context,
    private val onColorSelected: (r: Int, g: Int, b: Int) -> Unit
) : View(context) {

    private val wheelPaint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val pointerPaint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val pointerOutlinePaint = Paint(Paint.ANTI_ALIAS_FLAG)

    private var centerX = 0f
    private var centerY = 0f
    private var radius = 0f

    // Seçilen noktanın koordinatları
    private var pointerX = 0f
    private var pointerY = 0f

    init {
        pointerPaint.color = Color.WHITE
        pointerPaint.style = Paint.Style.FILL

        pointerOutlinePaint.color = ProColors.Background
        pointerOutlinePaint.style = Paint.Style.STROKE
        pointerOutlinePaint.strokeWidth = 6f

        minimumHeight = 500
    }

    override fun onSizeChanged(w: Int, h: Int, oldw: Int, oldh: Int) {
        super.onSizeChanged(w, h, oldw, oldh)
        centerX = w / 2f
        centerY = h / 2f
        radius = (Math.min(w, h) / 2f) * 0.85f // Biraz boşluk bırak

        // Hue (Renk) için 360 derecelik SweepGradient
        val colors = intArrayOf(
            Color.RED, Color.MAGENTA, Color.BLUE, Color.CYAN,
            Color.GREEN, Color.YELLOW, Color.RED
        )
        val sweepGradient = SweepGradient(centerX, centerY, colors, null)

        // Saturation (Doygunluk) için merkezden dışa beyazdan saydama RadialGradient
        val radialGradient = RadialGradient(
            centerX, centerY, radius,
            Color.WHITE, 0x00FFFFFF, Shader.TileMode.CLAMP
        )

        // İki gradient'i birleştir (Multiply modu ile)
        val composeShader = ComposeShader(sweepGradient, radialGradient, PorterDuff.Mode.SRC_OVER)
        wheelPaint.shader = composeShader

        // Başlangıç noktası (Merkez = Beyaz)
        pointerX = centerX
        pointerY = centerY
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        // Renk çarkını çiz
        canvas.drawCircle(centerX, centerY, radius, wheelPaint)

        // Seçici imleci (Pointer) çiz
        canvas.drawCircle(pointerX, pointerY, 24f, pointerPaint)
        canvas.drawCircle(pointerX, pointerY, 24f, pointerOutlinePaint)
    }

    @SuppressLint("ClickableViewAccessibility")
    override fun onTouchEvent(event: MotionEvent): Boolean {
        val x = event.x
        val y = event.y

        when (event.action) {
            MotionEvent.ACTION_DOWN, MotionEvent.ACTION_MOVE, MotionEvent.ACTION_UP -> {
                // Dokunulan noktanın merkeze uzaklığı ve açısı
                val dx = x - centerX
                val dy = y - centerY
                val distance = hypot(dx.toDouble(), dy.toDouble()).toFloat()

                // Çemberin dışına çıkmayı engelle
                if (distance <= radius) {
                    pointerX = x
                    pointerY = y
                } else {
                    val ratio = radius / distance
                    pointerX = centerX + dx * ratio
                    pointerY = centerY + dy * ratio
                }

                invalidate() // Çizimi güncelle

                // ACTION_UP (parmak çekildiğinde) rengi hesapla ve ESP'ye yolla
                if (event.action == MotionEvent.ACTION_UP) {
                    calculateColorAndTrigger()
                }
                return true
            }
        }
        return super.onTouchEvent(event)
    }

    private fun calculateColorAndTrigger() {
        val dx = pointerX - centerX
        val dy = pointerY - centerY
        val distance = hypot(dx.toDouble(), dy.toDouble()).toFloat()

        // Hue (0-360) hesapla
        var angle = Math.toDegrees(atan2(dy.toDouble(), dx.toDouble())).toFloat()
        if (angle < 0) angle += 360f

        // Saturation (0-1) hesapla
        val saturation = distance / radius

        // Value sabit 1.0 (Parlaklık master dim ile kontrol edilecek)
        val hsv = floatArrayOf(angle, saturation, 1f)
        val color = Color.HSVToColor(hsv)

        // 1023 formatına (ESP PWM formatı) çevir
        val r = (Color.red(color) / 255f * 1023).toInt()
        val g = (Color.green(color) / 255f * 1023).toInt()
        val b = (Color.blue(color) / 255f * 1023).toInt()

        // Seçilen rengi göstergeye ver
        pointerPaint.color = color
        invalidate()

        onColorSelected(r, g, b)
    }
}

class RadarScannerView(context: Context) : View(context) {

    private val radarPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = ProColors.Primary
        style = Paint.Style.STROKE
        strokeWidth = 4f
    }

    private val centerPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = ProColors.Primary
        style = Paint.Style.FILL
    }

    private var radiusProgress = 0f
    private var scanAnimator: ValueAnimator? = null

    init {
        // En az 300x300 piksel boyutlarında olmasını sağlıyoruz
        minimumWidth = 300
        minimumHeight = 300
    }

    fun startScanning() {
        if (scanAnimator?.isRunning == true) return

        scanAnimator = ValueAnimator.ofFloat(0f, 1f).apply {
            duration = 2000 // 2 saniyelik yavaş, profesyonel tarama efekti
            repeatCount = ValueAnimator.INFINITE
            interpolator = LinearInterpolator()
            addUpdateListener { animator ->
                radiusProgress = animator.animatedValue as Float
                invalidate() // Ekranı yeniden çiz
            }
            start()
        }
    }

    fun stopScanning() {
        scanAnimator?.cancel()
        radiusProgress = 0f
        invalidate()
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)

        val cx = width / 2f
        val cy = height / 2f
        val maxRadius = (Math.min(width, height) / 2f) * 0.9f

        // Merkez nokta
        canvas.drawCircle(cx, cy, 12f, centerPaint)

        // Büyüyen dalga 1
        radarPaint.alpha = ((1f - radiusProgress) * 255).toInt()
        canvas.drawCircle(cx, cy, maxRadius * radiusProgress, radarPaint)

        // Büyüyen dalga 2 (Yarım periyot geriden gelen ikinci dalga)
        val progress2 = (radiusProgress + 0.5f) % 1f
        radarPaint.alpha = ((1f - progress2) * 255).toInt()
        canvas.drawCircle(cx, cy, maxRadius * progress2, radarPaint)
    }

    override fun onDetachedFromWindow() {
        super.onDetachedFromWindow()
        stopScanning()
    }
}

package com.example.prolightmesh.storage

import android.content.Context
import org.json.JSONArray
import org.json.JSONObject
import android.util.Log
class MeshLocalStore(context: Context) {

    private val prefs = context.getSharedPreferences("pro_light_mesh_store", Context.MODE_PRIVATE)

    fun getGroups(): List<StoredGroup> {
        val raw = prefs.getString(KEY_GROUPS, null)

        if (raw.isNullOrBlank()) {
            return listOf(StoredGroup("Tüm Işıklar", DEFAULT_ALL_LIGHTS_GROUP))
        }

        val array = JSONArray(raw)
        val groups = mutableListOf<StoredGroup>()

        for (i in 0 until array.length()) {
            val obj = array.getJSONObject(i)
            groups.add(
                StoredGroup(
                    name = obj.getString("name"),
                    address = obj.getInt("address")
                )
            )
        }

        return groups
    }

    fun removeNodeFromGroup(nodeAddress: Int, groupAddress: Int) {
        val nodes = getNodes().toMutableList()
        val index = nodes.indexOfFirst { it.unicastAddress == nodeAddress }

        if (index < 0) return

        val node = nodes[index]

        nodes[index] = node.copy(
            groupAddresses = node.groupAddresses.filter { it != groupAddress }
        )

        saveNodes(nodes)
    }

    fun saveGroups(groups: List<StoredGroup>) {
        val array = JSONArray()

        groups.forEach { group ->
            array.put(
                JSONObject()
                    .put("name", group.name)
                    .put("address", group.address)
            )
        }

        prefs.edit().putString(KEY_GROUPS, array.toString()).apply()
    }

    fun addGroup(name: String): StoredGroup {
        val groups = getGroups().toMutableList()
        val group = StoredGroup(name, findNextGroupAddress(groups))

        groups.add(group)
        saveGroups(groups)

        return group
    }

    fun deleteGroup(address: Int) {
        if (address == DEFAULT_ALL_LIGHTS_GROUP) return

        saveGroups(getGroups().filter { it.address != address })

        val updatedNodes = getNodes().map { node ->
            node.copy(groupAddresses = node.groupAddresses.filter { it != address })
        }

        saveNodes(updatedNodes)
    }

    fun getNodes(): List<StoredNode> {
        val raw = prefs.getString(KEY_NODES, null) ?: return emptyList()
        val array = JSONArray(raw)
        val nodes = mutableListOf<StoredNode>()

        for (i in 0 until array.length()) {
            val obj = array.getJSONObject(i)
            val groupArray = obj.optJSONArray("groupAddresses") ?: JSONArray()

            val groups = mutableListOf<Int>()
            for (g in 0 until groupArray.length()) {
                groups.add(groupArray.getInt(g))
            }

            val macAddress = if (obj.has("macAddress") && !obj.isNull("macAddress")) {
                obj.optString("macAddress")
                    .takeIf { it.isNotBlank() && it != "null" }
            } else {
                null
            }

            nodes.add(
                StoredNode(
                    unicastAddress = obj.getInt("unicastAddress"),
                    name = obj.optString("name", "PRO-NODE"),
                    groupAddresses = groups,
                    macAddress = macAddress,

                    red = obj.optInt("red", 0),
                    green = obj.optInt("green", 0),
                    blue = obj.optInt("blue", 0),
                    masterDim = obj.optInt("masterDim", 0),

                    lastSeen = obj.optLong("lastSeen", 0L),
                    lastSeenByScan = obj.optLong(
                        "lastSeenByScan",
                        obj.optLong("lastSeen", 0L)
                    ),
                    lastSeenByMesh = obj.optLong("lastSeenByMesh", 0L),
                    lastRssi = obj.optInt("lastRssi", -100),

                    pendingDelete = obj.optBoolean("pendingDelete", false),
                    pendingDeleteAt = obj.optLong("pendingDeleteAt", 0L),
                    pendingResetLastAttemptAt = obj.optLong("pendingResetLastAttemptAt", 0L),
                    pendingResetAttemptCount = obj.optInt("pendingResetAttemptCount", 0)
                )
            )
        }

        return nodes
    }
    fun saveNodes(nodes: List<StoredNode>) {
        val array = JSONArray()

        nodes.forEach { node ->
            val groupArray = JSONArray()
            node.groupAddresses.forEach { groupArray.put(it) }

            val obj = JSONObject()
                .put("unicastAddress", node.unicastAddress)
                .put("name", node.name)
                .put("groupAddresses", groupArray)
                .put("red", node.red)
                .put("green", node.green)
                .put("blue", node.blue)
                .put("masterDim", node.masterDim)
                .put("lastSeen", node.lastSeen)
                .put("lastSeenByScan", node.lastSeenByScan)
                .put("lastSeenByMesh", node.lastSeenByMesh)
                .put("lastRssi", node.lastRssi)
                .put("pendingDelete", node.pendingDelete)
                .put("pendingDeleteAt", node.pendingDeleteAt)
                .put("pendingResetLastAttemptAt", node.pendingResetLastAttemptAt)
                .put("pendingResetAttemptCount", node.pendingResetAttemptCount)

            // 🔥 KRİTİK: null ise hiç yazma
            if (!node.macAddress.isNullOrBlank()) {
                obj.put("macAddress", node.macAddress)
            }

            array.put(obj)
        }

        prefs.edit().putString(KEY_NODES, array.toString()).apply()
    }

    fun upsertNode(node: StoredNode) {
        val nodes = getNodes().toMutableList()
        val index = nodes.indexOfFirst { it.unicastAddress == node.unicastAddress }

        if (index >= 0) {
            nodes[index] = node
        } else {
            nodes.add(node)
        }

        saveNodes(nodes)
    }

    fun addNodeToGroup(
        nodeAddress: Int,
        nodeName: String,
        groupAddress: Int,
        macAddress: String? = null
    ) {
        val nodes = getNodes().toMutableList()
        val index = nodes.indexOfFirst { it.unicastAddress == nodeAddress }

        if (index >= 0) {
            val node = nodes[index]
            val updatedGroups = (node.groupAddresses + groupAddress).distinct()

            nodes[index] = node.copy(
                name = nodeName,
                groupAddresses = updatedGroups,
                macAddress = macAddress ?: node.macAddress
            )
        } else {
            nodes.add(
                StoredNode(
                    unicastAddress = nodeAddress,
                    name = nodeName,
                    groupAddresses = listOf(groupAddress),
                    macAddress = macAddress
                )
            )
        }

        saveNodes(nodes)
    }

    fun removeNode(nodeAddress: Int) {
        saveNodes(getNodes().filter { it.unicastAddress != nodeAddress })
    }

    fun getNode(nodeAddress: Int): StoredNode? {
        return getNodes().firstOrNull { it.unicastAddress == nodeAddress }
    }

    // YENİ: Node statelerini tek bir harf komutuna göre güncelleyen fonksiyon
    fun updateNodeState(nodeAddress: Int, type: Char, value: Int) {
        val nodes = getNodes().toMutableList()
        val index = nodes.indexOfFirst { it.unicastAddress == nodeAddress }

        if (index < 0) return

        val node = nodes[index]
        val now = System.currentTimeMillis()

        val updatedNode = when (type) {
            'M' -> node.copy(masterDim = value, lastSeen = now, lastSeenByMesh = now)
            'R' -> node.copy(red = value, lastSeen = now, lastSeenByMesh = now)
            'G' -> node.copy(green = value, lastSeen = now, lastSeenByMesh = now)
            'B' -> node.copy(blue = value, lastSeen = now, lastSeenByMesh = now)
            'P' -> node.copy(lastSeen = now, lastSeenByMesh = now)
            else -> node.copy(lastSeen = now, lastSeenByMesh = now)
        }

        nodes[index] = updatedNode
        saveNodes(nodes)
    }
    fun updateLastSeen(nodeAddress: Int) {
        val nodes = getNodes().toMutableList()
        val index = nodes.indexOfFirst { it.unicastAddress == nodeAddress }

        if (index < 0) return

        nodes[index] = nodes[index].copy(
            lastSeen = System.currentTimeMillis()
        )

        saveNodes(nodes)
    }

    fun updateLastSeenByScanByMac(macAddress: String, rssi: Int) {
        val nodes = getNodes().toMutableList()

        val index = nodes.indexOfFirst {
            it.macAddress.equals(macAddress, ignoreCase = true)
        }

        if (index < 0) {

            Log.e(
                "SCAN_PRESENCE",
                "❌ MAC eşleşmedi incoming=$macAddress"
            )

            nodes.forEach {

                Log.e(
                    "SCAN_PRESENCE",
                    "KNOWN NODE addr=0x${it.unicastAddress.toString(16)} mac=${it.macAddress}"
                )
            }

            return
        }

        val node = nodes[index]
        val now = System.currentTimeMillis()

        // Aynı node çok sık gelirse store/UI spam olmasın
        if (now - node.lastSeenByScan < 1500L) {
            return
        }

        val rssiDiff = kotlin.math.abs(rssi - node.lastRssi)

        nodes[index] = node.copy(
            lastSeen = now,
            lastSeenByScan = now,

            // RSSI küçük değiştiyse eski RSSI kalsın, ama lastSeen güncellensin
            lastRssi = if (rssiDiff >= 5) rssi else node.lastRssi
        )

        Log.e(
            "SCAN_PRESENCE",
            "✅ SCAN UPDATE MAC addr=0x${node.unicastAddress.toString(16)} mac=$macAddress rssi=$rssi"
        )

        saveNodes(nodes)
    }

    fun markNodePendingDelete(nodeAddress: Int) {
        val nodes = getNodes().toMutableList()
        val index = nodes.indexOfFirst { it.unicastAddress == nodeAddress }

        if (index < 0) return

        val now = System.currentTimeMillis()

        nodes[index] = nodes[index].copy(
            pendingDelete = true,
            pendingDeleteAt = now
        )

        saveNodes(nodes)
    }

    fun getPendingDeleteNodeByMac(macAddress: String): StoredNode? {
        return getNodes().firstOrNull {
            it.pendingDelete &&
                    it.macAddress.equals(macAddress, ignoreCase = true)
        }
    }

    fun markPendingResetAttempt(nodeAddress: Int) {
        val nodes = getNodes().toMutableList()
        val index = nodes.indexOfFirst { it.unicastAddress == nodeAddress }

        if (index < 0) return

        val now = System.currentTimeMillis()
        val node = nodes[index]

        nodes[index] = node.copy(
            pendingResetLastAttemptAt = now,
            pendingResetAttemptCount = node.pendingResetAttemptCount + 1
        )

        saveNodes(nodes)
    }

    fun canAttemptPendingReset(nodeAddress: Int): Boolean {
        val node = getNode(nodeAddress) ?: return false
        val now = System.currentTimeMillis()

        if (!node.pendingDelete) return false

        // Aynı cihaza 30 saniyeden sık reset denemeyelim
        return now - node.pendingResetLastAttemptAt > 30_000L
    }

    fun debugBindKnownMacs() {
        val nodes = getNodes().toMutableList()

        for (i in nodes.indices) {
            nodes[i] = when (nodes[i].unicastAddress) {
                0x0002 -> nodes[i].copy(macAddress = "1C:DB:D4:5E:9A:36")
                0x0004 -> nodes[i].copy(macAddress = "3C:DC:75:6E:8B:FE")
                else -> nodes[i]
            }
        }

        saveNodes(nodes)
    }


    fun getMacByNodeAddress(nodeAddress: Int): String? {
        return getNodes()
            .firstOrNull { it.unicastAddress == nodeAddress }
            ?.macAddress
    }

    private fun findNextGroupAddress(groups: List<StoredGroup>): Int {
        var address = DEFAULT_ALL_LIGHTS_GROUP + 1
        val used = groups.map { it.address }.toSet()

        while (used.contains(address)) {
            address++
        }

        return address
    }

    companion object {
        const val DEFAULT_ALL_LIGHTS_GROUP = 0xC000

        private const val KEY_GROUPS = "groups"
        private const val KEY_NODES = "nodes"
    }
